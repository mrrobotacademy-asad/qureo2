مراجعة المصفوفات والتكرار والشروط المتداخلة

ابدأ بمراجعة نقاط الدرس وشرح الأكواد الأربعة خطوة بخطوة، ثم انتقل إلى أسئلة تتبع الخرج والتحدي العكسي وأسئلة المفاهيم. في النهاية احسب نتيجتك وجهّز تقريرك.

أولًا: نقاط شرح الدرس

ملخص المفاهيم الأساسية التي تحتاجها لفهم الأكواد.

1

المصفوفة Array

تخزن أكثر من قيمة داخل متغير واحد، وتكتب عناصرها بين الأقواس المربعة []، مع الفصل بين العناصر بالفواصل.

let volunteerList = ["Yes", "No", "Yes"];
2

حلقة for...of

تكرر تنفيذ الأوامر على عناصر المصفوفة، وفي كل دورة تأخذ قيمة العنصر الحالي.

for (let day of volunteerList) {
  // أوامر تتكرر لكل عنصر
}
3

الجملة الشرطية if

تختبر شرطًا؛ إذا كانت نتيجته true ينفذ البرنامج الأوامر داخل if.

if (day == "Yes") {
  count++;
}
4

عامل المقارنة ==

يقارن بين قيمتين للتحقق من تساويهما، وتكون نتيجة المقارنة true أو false.

5

العداد count++

يزيد قيمة المتغير بمقدار واحد. نبدأ غالبًا العداد من صفر ثم نزيده عند تحقق الشرط.

let count = 0;
count++;
6

الشروط المتداخلة Nested if

يمكن وضع if داخل if أخرى. لا نصل إلى فحص الشرط الداخلي إلا بعد تحقق الشرط الخارجي.

7

الجملة else

تنفذ عندما لا يتحقق الشرط المرتبط بها. الأقواس المعقوفة { } توضح بداية ونهاية كل جزء.

8

العامل المنطقي AND (&&)

يشترط تحقق الشرطين معًا حتى تكون نتيجة التعبير صحيحة.

if (weather == "sunny" && friends >= 10)

ثانيًا: الأكواد الأساسية ومعاينة الخرج

الأكواد الأربعة التي يدور حولها الشرح والاختبار.

الكود الأول والثاني: عدّ عناصر المصفوفة

let volunteerList = ["Yes", "No", "Yes", "Yes", "No", "Yes", "No", "Yes", "Yes", "No", "Yes", "Yes", "No", "Yes"];

let count = 0;
for (let day of volunteerList) {
  if (day == "Yes") {
    count++;
  }
}
console.log(count);
let count = 0;
for (let day of volunteerList) {
  if (day == "No") {
    count++;
  }
}
console.log(count);

الكود الثالث: الشروط المتداخلة

let weather = "sunny";
let friends = 10;

if (weather == "sunny") {
  if (friends >= 10) {
    console.log("play baseball");
  } else {
    console.log("play basketball");
  }
} else {
  console.log("read a book");
}

if (sandwichType == "savory") {

2. التحدي العكسي (Output Matching)

اقرأ الخرج المطلوب واختر الكود الذي ينتجه.

3. أسئلة المفاهيم وتراكيب الكود

اختبر فهمك للمصفوفات والحلقات والشروط.
